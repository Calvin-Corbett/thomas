1. BASIC PUBLISH/SUBSCRIBE

    from thomas.event_bus import Event, EventBusContext

    class UserCreated(Event):
        def __init__(self, user_id: str, email: str):
            super().__init__('UserCreated', f'user-{user_id}')
            self.user_id = user_id
            self.email = email

    async def handle_user_created(event: UserCreated):
        print(f"User created: {event.email}")

    async with EventBusContext() as bus:
        bus.subscribe(handle_user_created, event_types={'UserCreated'})
        await bus.publish(UserCreated('123', 'user@example.com'))

2. EVENT SOURCING WITH AGGREGATES

    from thomas.event_bus import AggregateRoot, AggregateRepository, EventStore

    class UserAggregate(AggregateRoot):
        def _on_user_created(self, event: UserCreated):
            self._state['email'] = event.email

        def create_user(self, email: str):
            self.record_event(UserCreated(self.id, email))

    store = EventStore()
    repo = AggregateRepository(store)

    user = UserAggregate('user-123')
    user.create_user('user@example.com')
    repo.save(user)

3. CQRS WITH PROJECTIONS

    from thomas.event_bus import InMemoryProjection, ProjectionManager

    class UserCountProjection(InMemoryProjection):
        def __init__(self):
            super().__init__('user_count', {'count': 0})

        async def _on_user_created(self, event):
            self._state['count'] += 1

    manager = ProjectionManager(store)
    projection = UserCountProjection()
    manager.register_projection(projection)

    await manager.catchup_all()
    print(projection.get_state())  # {'count': 2}

4. SAGAS FOR WORKFLOWS

    from thomas.event_bus import SagaBuilder, SagaExecutor

            .add_step('reserve_inventory', reserve_fn, unreserve_fn)
            .build())
